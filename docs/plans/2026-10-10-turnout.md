# Turnout section implementation plan

> **For Claude:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task.

**Goal:** A Turnout section that tracks Maryland election campaigns the way the SBE publishes them, starting with the 2026 general election's mail-in ballot tracking.

**Architecture:** A new election-keyed pipeline family (`pipeline/turnout.py`) downloads and snapshots the press-room stat files, parses them with strict reconciliation, and writes CSVs + provenance sidecars under `data/turnout/`; `build-db` loads two new tables; a `/turnout/` section (routers, templates, chart JS, static-site entries, downloads) renders them; a scheduled GitHub Action snapshots daily.

**Tech Stack:** existing pipeline idioms (httpx, natural-pdf, polars, provenance sidecars), new dep `python-calamine` for XLSX, existing chart helpers (`charthelpers.js`, `charts.css`).

**Design doc:** `docs/plans/2026-10-10-turnout-design.md` (read it first; it explains the SBE report families and the term definitions).

Two amendments to the design doc are folded into Task 1 (they came from reading the real files after the doc was written):

- The requests report is parsed from its **PDF only** — the live xlsx has corrupt zip members on SBE's side and its correct shape is unknown, so there is nothing to parse or test against. The sent-returned family is parsed from its **xlsx** (all three geography files read cleanly today). If an xlsx is ever unreadable, the parse fails loudly (red CI) rather than silently skipping, except the requests fallback noted above.
- The same-day county xlsx and CD xlsx printed different totals for the same as-of date, so reconciliation is strictly **within one document** (each table's own printed totals); never cross-check the PDF against the xlsx or one geography against another.

```python
import polars as pl

COUNTY_FIXES = {"Saint Mary'S": "St. Mary's", "Saint Mary's": "St. Mary's"}
N_COUNTIES = 24

REQUESTS_METHODS = ["agent", "faxed", "in-person", "mailed", "others", "print-at-home", "total"]


class TurnoutParseError(RuntimeError):
    """A turnout file did not parse with its own printed totals."""


REQUESTS_ROW_RE = re.compile(
    r"^(?P<county>[A-Z][A-Za-z' .]+?)\s+(?P<numbers>(?:\d[\d,]*\s+){6}\d[\d,]*)\s*$"
)


def canonical_county(name: str) -> str:
    name = name.strip()
    return COUNTY_FIXES.get(name, name.title())


def parse_requests_text(text: str) -> tuple[str, pl.DataFrame]:
    """The county x delivery-method request table from one requests PDF's text.

    Returns the as-of date and one row per county x measure (party is ALL).
    """
    as_of = extract_as_of_text(text)
    if as_of is None:
        raise TurnoutParseError("No 'As of:' line in the requests report")
    rows = []
    total = None
    for line in text.splitlines():
        line = line.strip()
        if line.startswith("TOTAL"):
            nums = [int(n.replace(",", "")) for n in line.split()[1:]]
            if len(nums) != 7:
                raise TurnoutParseError(f"Unexpected TOTAL row in requests report: {line!r}")
            total = nums
            continue
        if m := REQUESTS_ROW_RE.match(line):
            county, nums = canonical_county(m["county"]), [
                int(n.replace(",", "")) for n in m["numbers"].split()
            ]
            if len(nums) != 7:
                raise TurnoutParseError(f"Unexpected county row in requests report: {line!r}")
            if nums[-1] != sum(nums[:-1]):
                raise TurnoutParseError(f"County total does not add up for {county}: {nums}")
            for measure, value in zip(REQUESTS_METHODS, nums, strict=True):
                rows.append(
                    {
                        "as_of": as_of,
                        "election": "GG26",  # overwrriten by the caller? no — see below
                        "family": "requests",
                        "geography": "county",
                        "county": county,
                        "district": "",
                        "category": "ALL",
                        "party": "ALL",
                        "measure": measure,
                        "value": value,
                    }
                )
    if total is None or len(rows) == 0:
        raise TurnoutParseError("Requests report has no county rows or no TOTAL row")
    if total[-1] != sum(rows[i]["value"] for i in range(0, len(rows), 7)):
        raise TurnoutParseError("County totals do not sum to the statewide TOTAL")
    return as_of, pl.DataFrame(rows)
```

The three xlsx files share a shape: a header row containing `CATEGORY`, then either `COUNTY NAME` before `DISTRICT` (county file Sheet1: `DISTRICT` = `Statewide`; congressional/legislative Sheet1s swap `DISTRICT` first with `COUNTY NAME` = `ALL`), spacer blank columns, label variants (`REP SENT` vs `REP_SENT`, `OTHR ACCEPTED` vs `OTH ACCEPTED`), blocks of rows per voter category, per-block `Total` rows, and a final `Grand Total`.

Parse rule (header-driven, from the repo's own philosophy): read the header row, map every non-empty label to a `(party, measure)` pair, then read every data row below it; a row's identity is `(category, district, county)`.

```python
from skipjack.pipeline.fetch import (

from skipjack.pipeline.provenance import SourceRecord


def download(url: str, timeout: float = 30.0) -> bytes:
    """Download a press-room file; SBE answers 200 with an HTML page for unpublished files."""
    import httpx

    resp = httpx.get(url, timeout=timeout, follow_redirects=True)
    resp.raise_for_status()
    return resp.content


def sniff(data: bytes) -> str | None:
    if data.startswith(b"%PDF"):
        return "pdf"
    if data.startswith(b"PK"):
        return "xlsx"
    return None


def turnout_fetch(force: bool = False, election: str | None = None) -> list[SourceRecord]:
    """Download every configured turnout file whose content changed; archive and record it."""
    fetched = []
    for e in (ELECTIONS.values() if election is None else [ELECTIONS[election]]):
        for f in e.files:
            data = download(f.url)
            kind = sniff(data)
            if kind is None:
                continue  # not published yet
            as_of = extract_as_of_bytes(data, kind)
            sha = hashlib.sha256(data).hexdigest()
            if not force and _latest_sha(e.code, f.family, f.geography, kind) == sha:
                continue
            stamp = datetime.datetime.now().strftime("%Y%m%dT%H%M%S")
            dest = (
                ARCHIVES_DIR / "sbe" / "campaign_stats" / e.code
                / f"{f.family}__{f.geography}__{as_of or 'unknown'}__{stamp}.{kind}"
            )
            dest.parent.mkdir(parents=True, exist_ok=True)
            dest.write_bytes(data)
            source_id = f"sbe-turnout-{e.code.lower()}-{f.family}-{f.geography}-{(as_of or 'unknown').replace('-', '')}-{stamp}"
            record = SourceRecord.create(
                source_id=source_id,
                source_url=f.url,
                source_name=SOURCE_NAME,
                local_path=dest,
                document_title=f"{e.code} {f.family} ({f.geography}) as of {as_of or 'unknown'}",
            )
            append_to_manifest(MANIFEST_PATH, record)
            fetched.append(record)
    return fetched


def _latest_sha(election: str, family: str, geography: str, kind: str) -> str | None:
    """SHA of the newest archived revision of this file, from the manifest order."""
    prefix = f"sbe-turnout-{election.lower()}-{family}-{geography}-"
    for row in reversed(read_manifest(MANIFEST_PATH)):
        if row.source_id.startswith(prefix) and row.local_path.endswith(kind):
            return row.sha256
    return None


def extract_as_of_bytes(data: bytes, kind: str) -> str | None:
    if kind == "xlsx":
        import tempfile

        with tempfile.NamedTemporaryFile(suffix=".xlsx") as fh:
            fh.write(data)
            fh.flush()
            return extract_as_of_xlsx(Path(fh.name))
    return extract_as_of_text(data.decode("latin-1", errors="replace"))
```
